\section{Results Appendix}

\subsection{Fidelity}
The fidelity between reconstructed density matrix $\rho$ and true state $\sigma$ is:

\begin{equation}
F(\rho,\sigma)=\left(\mathrm{Tr}\sqrt{\sqrt{\rho}\sigma\sqrt{\rho}}\right)^2
\end{equation}

\subsection{Trace Distance}
\begin{equation}
D(\rho,\sigma)=\frac{1}{2}\|\rho-\sigma\|_1
\end{equation}
